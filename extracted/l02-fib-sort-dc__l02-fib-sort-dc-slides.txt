function GmInt2(n; v1 = v1, v2 = v2, a1 = a1, a2 = a2,
l1 = l1, l2 = l2)
b1 = l1 * a1 ^ (n-1)
b2 = l2 * a2 ^ (n-1)
fn = round(Int64, b1 * v1[1] + b2 * v2[1])
fn2 = round(Int64, b1 * v1[2] + b2 * v2[2])
return fn, fn2
end

function bubble_sort!(v::AbstractVector)
tmp = zero(eltype(v))
len = length(v)
for i in 1:len
for j in 1:len-i
if v[j] > v[j+1]
tmp = v[j]
v[j] = v[j+1]
v[j+1] = tmp
end
end
end
return nothing
end

function insertion_sort!
(v::AbstractVector)
len = length(v)
for i in 2:len
j = i
x = v[i]
while j > 1
y = v[j-1]
if x > y
break
end
v[j] = y
j -= 1
end
v[j] = x
end
return v
end

function quicksort!(v::AbstractVector, lo, hi, t::AbstractVector)
if hi - lo <= 16
vv = @view v[lo:hi]

end

lo1, hi1, lo2, hi2 = partition!(v, lo, hi, t)
if lo1 < hi1
quicksort!(v, lo1, hi1, t)
end
if lo2 < hi2
quicksort!(v, lo2, hi2, t)
end
return nothing
end

function partition!(v::AbstractVector, lo, hi, t::AbstractVector)
ngreater = zero(lo)
nequal = zero(lo)
myone = one(lo)
pivot_index = rand(lo:hi) # 随机取枢轴，避免在已排序输入上退化
pivot = v[pivot_index]

@inbounds begin
for i in lo:hi
x = v[i]
if x < pivot
t[i-ngreater-nequal] = x
elseif x > pivot
t[hi-ngreater] = x
ngreater += myone
else
nequal += myone
end
end
t[hi-ngreater-nequal+myone:hi-ngreater] .= pivot
