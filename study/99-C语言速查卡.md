## 1. 动态数组 `Vec`（★ 必背）

```c
typedef struct { int *data; size_t size, cap; } Vec;

static int vec_reserve(Vec *v, size_t need) {
    if (need <= v->cap) return 1;
    size_t nc = v->cap ? v->cap : 2;
    while (nc < need) nc *= 2;                          /* 翻倍，均摊 O(1) */
    int *tmp = (int *)realloc(v->data, nc * sizeof(int));
    if (!tmp) return 0;                                 /* 失败：旧内存原样保留 */
    v->data = tmp; v->cap = nc;
    return 1;
}
static int vec_push(Vec *v, int x) {
    if (!vec_reserve(v, v->size + 1)) return 0;
    v->data[v->size++] = x;
    return 1;
}
static void vec_free(Vec *v) { free(v->data); v->data = NULL; v->size = v->cap = 0; }
```

**实测**：`for (i=1..10) vec_push(&v, i*i);` → `size=10 cap=16 data[9]=100`

**必答三问**：
1. **为什么不每次扩容 1 个？** 总搬运量 1+2+…+n = O(n²)。
2. **为什么翻倍就够？** 总搬运量 1+2+4+…< 2n = O(n)，均摊每次 O(1)。
3. **为什么不能写 `v->data = realloc(v->data, n)`？** 失败返回 `NULL` 会覆盖原指针，旧内存永久丢失（泄漏）。

## 2. 单链表：节点与两种插入（★ 必背）

```c
typedef struct Node { int val; struct Node *next; } Node;

static Node *node_new(int v) {
    Node *n = (Node *)malloc(sizeof *n);
    if (!n) return NULL;
    n->val = v; n->next = NULL;
    return n;
}

/* 头插 O(1)：返回新的头 */
static Node *push_front(Node *head, int v) {
    Node *n = node_new(v);
    if (!n) return head;
    n->next = head;          /* ① 先接后继 */
    return n;                /* ② 再改头 */
}

/* 尾插 O(n)：返回头（空表时头会变，所以必须返回） */
static Node *push_back(Node *head, int v) {
    Node *n = node_new(v);
    if (!n) return head;
    if (!head) return n;
    Node *p = head;
    while (p->next) p = p->next;
    p->next = n;
    return head;
}

static void list_print(const Node *h) {
    putchar('[');
    for (const Node *p = h; p; p = p->next)
        printf(p->next ? "%d -> " : "%d", p->val);
    puts("]");
}

static void list_free(Node *h) {          /* 先存后继，再 free —— 顺序不能反 */
    while (h) { Node *nx = h->next; free(h); h = nx; }
}
```

**实测**：`push_back 1..5` 再 `push_front 0` → `[0 -> 1 -> 2 -> 3 -> 4 -> 5]`

**为什么头插三步不能颠倒**：若先写 `head = n;` 再写 `n->next = head;`，`n->next` 就指向自己（自环），原来的整条链**永久丢失**。
**口诀**：**先接后继，再改前驱。**

## 3. 按值删除（双指针 / 二级指针法）（★ 必背）

```c
static Node *delete_value(Node *head, int v) {
    Node **pp = &head;                                  /* 指向"存当前结点指针的位置" */
    while (*pp && (*pp)->val != v) pp = &(*pp)->next;
    if (*pp) { Node *dead = *pp; *pp = dead->next; free(dead); }
    return head;
}
```

**实测**：`[0..5]` 删 3 → `[0 -> 1 -> 2 -> 4 -> 5]`

**为什么用二级指针**：`pp` 一开始指向 `head` 变量本身，之后指向某个 `->next` 字段。这样"删除头结点"和"删除中间结点"是**同一行代码**，不需要特判。

**等价写法（哨兵/哑结点法）**：

```c
static Node *delete_value2(Node *head, int v) {
    Node dummy = {0, head};          /* 栈上哑结点，next 指向真头 */
    Node *prev = &dummy;
    while (prev->next && prev->next->val != v) prev = prev->next;
    if (prev->next) { Node *dead = prev->next; prev->next = dead->next; free(dead); }
    return dummy.next;
}
```

## 4. 按位置插入 / 删除（★ 必背）

```c
static Node *insert_at(Node *head, size_t pos, int v) {
    if (pos == 0) return push_front(head, v);
    Node *p = head;
    for (size_t i = 0; p && i + 1 < pos; i++) p = p->next;   /* 走到 pos-1 */
    if (!p) return head;                                     /* pos 越界 */
    Node *n = node_new(v);
    if (!n) return head;
    n->next = p->next;                                       /* ① 新结点接后继 */
    p->next = n;                                             /* ② 前驱接新结点 */
    return head;
}

static Node *delete_at(Node *head, size_t pos) {
    if (!head) return NULL;
    if (pos == 0) { Node *nx = head->next; free(head); return nx; }
    Node *p = head;
    for (size_t i = 0; p->next && i + 1 < pos; i++) p = p->next;
    if (!p->next) return head;                               /* pos 越界 */
    Node *dead = p->next;
    p->next = dead->next;                                    /* ① 先接后继 */
    free(dead);                                              /* ② 再释放 */
    return head;
}
```

**实测**：位置 2 插 99 → `[0 -> 1 -> 99 -> 2 -> 4 -> 5]`；位置 2 删 → `[0 -> 1 -> 2 -> 4 -> 5]`

**三条边界必须处理**：空表、`pos == 0`（改头）、`pos` 越界（不改动并返回原头）。

## 5. 反转链表：迭代三指针（★ 必背）

```c
static Node *reverse(Node *head) {
    Node *prev = NULL, *cur = head;
    while (cur) {
        Node *nxt = cur->next;      /* ① 存后继 */
        cur->next = prev;           /* ② 掉头 */
        prev = cur;                 /* ③ prev 前移 */
        cur = nxt;                  /* ④ cur 前移 */
    }
    return prev;                    /* prev 停在原尾，即新头 */
}
```

**实测**：`[0 -> 1 -> 2 -> 4 -> 5]` → `[5 -> 4 -> 2 -> 1 -> 0]`，再反转复原。

**画图法（考试时在草稿纸画这四步）**：

```
初始:   prev=NULL   cur=①→②→③→NULL
第1轮:  nxt=②  ①->next=NULL  prev=①  cur=②     链: NULL←①  ②→③→NULL
第2轮:  nxt=③  ②->next=①     prev=②  cur=③     链: NULL←①←②  ③→NULL
第3轮:  nxt=NULL ③->next=②    prev=③  cur=NULL   链: NULL←①←②←③
返回 prev = ③
```

**递归版（面试常问，注意 O(n) 递归深度会爆栈）**：

```c
static Node *reverse_rec(Node *head) {
    if (!head || !head->next) return head;
    Node *new_head = reverse_rec(head->next);
    head->next->next = head;        /* 让后继反过来指向自己 */
    head->next = NULL;              /* 自己变成尾 */
    return new_head;
}
```

## 6. 快慢指针四件套（★ 必背）

```c
/* (1) 找中点：偶数个返回"中间偏右" */
static Node *middle(Node *head) {
    Node *slow = head, *fast = head;
    while (fast && fast->next) { slow = slow->next; fast = fast->next->next; }
    return slow;
}

/* (2) 倒数第 k 个（k 从 1 开始）：先让 a 走 k 步，再一起走 */
static Node *kth_from_end(Node *head, size_t k) {
    if (k == 0) return NULL;
    Node *a = head, *b = head;
    for (size_t i = 0; i < k; i++) { if (!a) return NULL; a = a->next; }
    while (a) { a = a->next; b = b->next; }
    return b;
}

/* (3) 检环（Floyd 判圈）：返回相遇点，无环 NULL */
static Node *has_cycle(Node *head) {
    Node *slow = head, *fast = head;
    while (fast && fast->next) {
        slow = slow->next; fast = fast->next->next;
        if (slow == fast) return slow;
    }
    return NULL;
}

/* (4) 环入口：一个指针从头、一个从相遇点，同速走，相遇处即入口 */
static Node *cycle_entry(Node *head) {
    Node *meet = has_cycle(head);
    if (!meet) return NULL;
    while (head != meet) { head = head->next; meet = meet->next; }
    return head;
}
```

**实测**：链 `1..6` → 中点 `val=4`；倒数第 2 → `val=5`；倒数第 9 → `NULL`；无环 → 无；尾接第 3 个造环 → 有环且入口 `val=3`。

**两个为什么（考试爱问）**

1. **为什么快指针走两步、慢指针走一步一定相遇？**
   设入环前有 $a$ 步。慢指针入环时，快指针已在环内。之后两者每轮距离变化 $2-1=1$，环长有限（设为 $L$），故最多 $L$ 轮内距离必为 0（或说相对速度 1 必然追上）。
2. **为什么"一个从头、一个从相遇点同速走"会在环入口相遇？**
   设头到入口 $a$，入口到相遇点 $b$，相遇点绕回入口 $c$，环长 $L=b+c$。
   相遇时慢走 $a+b$，快走 $a+b+nL$，而快走的是慢的 2 倍：$2(a+b)=a+b+nL \Rightarrow a+b=nL \Rightarrow a = nL-b = (n-1)L + c$。
   所以从头走 $a$ 步、从相遇点走 $c$ 步（再加上整数圈）会同时到达入口。

**注意**：`while (fast && fast->next)` 的两个条件都不能少——`fast` 为 `NULL` 时读 `fast->next` 会崩。

## 7. 合并两条有序链表（★ 必背）

```c
static Node *merge_sorted(Node *a, Node *b) {
    Node dummy;                       /* 栈上哑结点：不用特判"结果为空" */
    Node *tail = &dummy;
    dummy.next = NULL;
    while (a && b) {
        if (a->val <= b->val) { tail->next = a; a = a->next; }   /* <= 保证稳定 */
        else                  { tail->next = b; b = b->next; }
        tail = tail->next;
    }
    tail->next = a ? a : b;           /* 接上剩余整段（O(1)，无需逐个搬） */
    return dummy.next;
}
```

**实测**：`[1,3,5]` 与 `[2,4,6]` → `[1 -> 2 -> 3 -> 4 -> 5 -> 6]`

**注意**：这个版本**原地改指针**，参数链表会被破坏，合并后 `a`/`b` 不再可用。若要保护输入，必须新建结点（`code/cproject` 的 `list_merge_sorted` 就是拷贝值版本）。

## 8. 三种数组排序（★ 必背）

```c
/* 冒泡：稳定，最好 O(n)（带 swapped 优化），最坏/平均 O(n^2) */
static void bubble_sort(int *a, int n) {
    for (int i = 0; i < n - 1; i++) {
        int swapped = 0;
        for (int j = 0; j < n - 1 - i; j++)
            if (a[j] > a[j + 1]) { int t = a[j]; a[j] = a[j + 1]; a[j + 1] = t; swapped = 1; }
        if (!swapped) break;
    }
}

/* 插入：稳定，最好 O(n)（近乎有序时极快），最坏/平均 O(n^2) */
static void insertion_sort(int *a, int n) {
    for (int i = 1; i < n; i++) {
        int key = a[i], j = i - 1;
        while (j >= 0 && a[j] > key) { a[j + 1] = a[j]; j--; }
        a[j + 1] = key;
    }
}

/* 快速：不稳定，平均 O(n log n)，最坏 O(n^2)；三数取中避免有序输入退化 */
static void quick_sort(int *a, int lo, int hi) {
    if (lo >= hi) return;
    int mid = lo + (hi - lo) / 2;
    if (a[mid] < a[lo]) { int t = a[mid]; a[mid] = a[lo]; a[lo] = t; }
    if (a[hi]  < a[lo]) { int t = a[hi];  a[hi]  = a[lo]; a[lo] = t; }
    if (a[hi]  < a[mid]){ int t = a[hi];  a[hi]  = a[mid]; a[mid] = t; }
    int pivot = a[mid];
    int i = lo, j = hi;
    while (i <= j) {
        while (a[i] < pivot) i++;
        while (a[j] > pivot) j--;
        if (i <= j) { int t = a[i]; a[i] = a[j]; a[j] = t; i++; j--; }
    }
    quick_sort(a, lo, j);
    quick_sort(a, i, hi);
}
```

**实测**：`{5,2,9,1,5,6,0,3,5}` → 三者均得 `0 1 2 3 5 5 5 6 9`

| 算法 | 最好 | 平均 | 最坏 | 空间 | 稳定 |
|---|---|---|---|---|---|
| 冒泡 | O(n) | O(n²) | O(n²) | O(1) | ✅ |
| 插入 | O(n) | O(n²) | O(n²) | O(1) | ✅ |
| 快速 | O(n log n) | O(n log n) | O(n²) | O(log n) 栈 | ❌ |
| 归并 | O(n log n) | O(n log n) | O(n log n) | O(n) | ✅ |

## 9. 归并排序（数组）（★ 必背）

```c
static void merge_range(int *a, int *tmp, int lo, int mid, int hi) {
    int i = lo, j = mid + 1, k = lo;
    while (i <= mid && j <= hi) tmp[k++] = (a[i] <= a[j]) ? a[i++] : a[j++];  /* <= 保稳定 */
    while (i <= mid) tmp[k++] = a[i++];
    while (j <= hi)  tmp[k++] = a[j++];
    for (int t = lo; t <= hi; t++) a[t] = tmp[t];
}

static void merge_sort_rec(int *a, int *tmp, int lo, int hi) {
    if (lo >= hi) return;
    int mid = lo + (hi - lo) / 2;
    merge_sort_rec(a, tmp, lo, mid);
    merge_sort_rec(a, tmp, mid + 1, hi);
    merge_range(a, tmp, lo, mid, hi);
}

static int merge_sort(int *a, int n) {          /* 返回 0 表示分配失败 */
    if (n <= 1) return 1;
    int *tmp = (int *)malloc((size_t)n * sizeof(int));
    if (!tmp) return 0;
    merge_sort_rec(a, tmp, 0, n - 1);
    free(tmp);
    return 1;
}
```

**为什么 `mid = lo + (hi - lo) / 2`**：避免 `lo + hi` 在极大下标时溢出。

## 10. 链表归并排序（★ 必背，L04 核心）

```c
static Node *split_middle(Node *head) {          /* 断成两半，返回后半的头 */
    Node *slow = head, *fast = head, *prev = NULL;
    while (fast && fast->next) { prev = slow; slow = slow->next; fast = fast->next->next; }
    if (prev) prev->next = NULL;                 /* 关键：断开，否则无限递归 */
    return slow;
}

static Node *list_merge_sort(Node *head) {
    if (!head || !head->next) return head;       /* 0 或 1 个结点已有序 */
    Node *mid = split_middle(head);
    Node *l = list_merge_sort(head);
    Node *r = list_merge_sort(mid);
    return merge_sorted(l, r);                   /* 复用第 7 节的合并 */
}
```

**实测**：`5 2 9 1 5 6 0 3 5` → `[0 -> 1 -> 2 -> 3 -> 5 -> 5 -> 5 -> 6 -> 9]`

**为什么链表归并排序比数组更自然**：合并阶段只需**改指针**，不需要额外 O(n) 的辅助数组——空间是 O(log n)（递归栈）而不是 O(n)。

## 11. 结构体 + 动态内存：成对的 create / destroy（★ 必背）

```c
typedef struct { int id; char *name; float score; } Student;

static Student *student_create(int id, const char *name, float score) {
    Student *s = (Student *)malloc(sizeof *s);
    if (!s) return NULL;
    s->id = id; s->score = score;
    s->name = (char *)malloc(strlen(name) + 1);      /* +1 放 '\0' */
    if (!s->name) { free(s); return NULL; }          /* 半成品要自己清干净 */
    strcpy(s->name, name);
    return s;
}

static void student_destroy(Student *s) {
    if (!s) return;
    free(s->name);      /* ① 先释放内部成员 */
    free(s);            /* ② 再释放结构体本身 */
}
```

**实测**：`202501 张三 95.50`

**为什么顺序不能反**：`free(s)` 之后 `s->name` 这个指针值就读不到了，字符串块永久泄漏。

## 12. 函数指针与 `qsort`（★ 必背）

```c
static int cmp_asc(const void *a, const void *b) {
    int x = *(const int *)a, y = *(const int *)b;
    return (x > y) - (x < y);          /* 不要写 x - y：大数相减会溢出 */
}
static int cmp_desc(const void *a, const void *b) { return cmp_asc(b, a); }

int a[6] = {3, 1, 4, 1, 5, 9};
qsort(a, 6, sizeof(int), cmp_asc);     /* 升序 */
```

**实测**：升序 `1 1 3 4 5 9`；降序 `9 5 4 3 1 1`

```c
typedef int (*CmpFn)(const void *, const void *);

static void my_sort(void *base, size_t n, size_t width, CmpFn cmp) { /* ... */ }
```

**函数指针的声明读法**：`int (*f)(int, int)` —— `f` 是指针，指向"接受两个 `int`、返回 `int`"的函数。

## 13. 文件读写（★ 必背，L04 Gantt 作业要用）

```c
static int file_roundtrip(const char *path, const int *a, int n, int *out) {
    FILE *f = fopen(path, "w");
    if (!f) return 0;
    for (int i = 0; i < n; i++)
        if (fprintf(f, "%d\n", a[i]) < 0) { fclose(f); return 0; }   /* 每次写都要查 */
    if (fclose(f) != 0) return 0;                                    /* fclose 也会失败 */

    f = fopen(path, "r");
    if (!f) return 0;
    int cnt = 0;
    while (cnt < n && fscanf(f, "%d", &out[cnt]) == 1) cnt++;        /* 用返回值判断 */
    fclose(f);
    return cnt;                                                      /* 返回真正读到的个数 */
}
```

**实测**：写入 9 个、读回 9 个、首元素 5，排序后 `0 1 2 3 5 5 5 6 9`

**四条铁律**：
1. `fopen` 返回 `NULL` 必须处理（文件不存在/无权限）；
2. `fscanf` 用**返回值**判断是否读成功，**不要**用 `while (!feof(f))`（会多读一次）；
3. `fclose` 也要检查返回值（写入可能因磁盘满而失败）；
4. 每个 `fopen` 都要有配对的 `fclose`，包括出错分支。

## 14. 段错误 / 结果不对时的 60 秒排查法

**常见症状 → 病因速查**

| 症状 | 最可能的原因 |
|---|---|
| `Segmentation fault` 且 `p` 是 `NULL` 或极小值 | 未初始化指针 / `malloc` 失败没检查 / 已 `free` 的悬空指针 |
| `Segmentation fault` 且地址看起来正常 | 越界写破坏了相邻内存 |
| `double free or corruption` | 同一指针释放两次；或改写了堆块头部（越界写） |
| 输出随机数 | 读了未初始化的 `malloc` 内存 |
| 结果时对时错 | 用了未初始化变量；或存在越界写 |
| 递归很深就崩 | 递归爆栈（如 `reverse_rec` 用于十万级链表） |
| 读到旧的/错的数据 | 忘了 `free` 后又继续用；或浅拷贝共享了指针 |
| `malloc(): corrupted top size` | 越界写把堆元数据写坏了，**真正的 bug 不在这里** |

## 15. 考前 30 分钟速记清单

```
□ malloc → 判空 → 初始化 → 用 → free → 置 NULL
□ 链表插入：先接后继，再改前驱
□ 链表删除：先存后继，再 free
□ 反转：存后继 → 掉头 → prev 前移 → cur 前移
□ 检环：慢 1 快 2，相遇即环；一个回头、一个留相遇点，同速走即入口
□ 倒数第 k：先走 k 步，再同步走
□ 归并：哑结点 + 尾指针 + 最后接剩余整段
□ 快排：三数取中 + Hoare 划分 + 只递归较短一侧（可选优化）
□ 归并排序：mid = lo + (hi-lo)/2；合并用 <= 保稳定
□ 链表归并：找中点时"断链"那一步绝对不能忘
□ 结构体释放：先 free 内部成员，再 free 结构体
□ 文件：fopen 判空、fscanf 看返回值、fclose 也检查
□ 编译：-Wall -Wextra -g -fsanitize=address,undefined
```
